========================================
RÈGLES DE VALIDATION
========================================

EMAIL:
- Format valide (xxx@xxx.xxx)
- Converti en minuscules

MOT DE PASSE:
- Minimum 8 caractères
- 1 majuscule minimum
- 1 minuscule minimum
- 1 chiffre minimum

TÉLÉPHONE (Cameroun):
- Format: 6XXXXXXXX
- 9 chiffres commençant par 6

PRIX:
- Minimum: 100 FCFA
- Maximum: 10,000,000 FCFA

RÉDUCTION:
- Minimum: 5%
- Maximum: 90%

STOCK:
- Minimum: 1
- Maximum: 10,000

========================================
EXEMPLE D'UTILISATION
========================================

import { useForm } from '@/lib/hooks/useForm';
import { loginSchema } from '@/lib/validations/schemas';
import FormInput from '@/components/ui/FormInput';

const {
  values,
  errors,
  isSubmitting,
  handleChange,
  handleSubmit,
  validateField
} = useForm({
  schema: loginSchema,
  onSubmit: async (data) => {
    // Logique de connexion
  },
  initialValues: {
    email: '',
    password: ''
  }
});

<FormInput
  label="Email"
  type="email"
  value={values.email || ''}
  onChange={(value) => handleChange('email', value)}
  onBlur={() => validateField('email')}
  error={errors.email}
  required
/>
